cmake_minimum_required(VERSION 3.20)
project(NekoExtraHUD VERSION 0.1.15 LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Use a Microsoft-ABI x86 toolchain: cmake -A Win32")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(NekoExtraHUD SHARED src/plugin.cpp src/runtime.cpp src/layout.cpp src/renderer.cpp NekoExtraHUD.def)
target_include_directories(NekoExtraHUD PRIVATE src)
target_compile_definitions(NekoExtraHUD PRIVATE NOMINMAX)
target_compile_options(NekoExtraHUD PRIVATE /W4 /permissive-)
target_link_libraries(NekoExtraHUD PRIVATE bcrypt user32)
add_executable(layout_tests tests/layout_tests.cpp src/layout.cpp)
target_include_directories(layout_tests PRIVATE src)
enable_testing()
add_test(NAME layout_rules COMMAND layout_tests)
add_executable(paint_context_tests tests/paint_context_tests.cpp)
target_include_directories(paint_context_tests PRIVATE src)
add_test(NAME paint_context_cleanup COMMAND paint_context_tests)
add_executable(runtime_adapter_tests tests/runtime_adapter_tests.cpp)
target_include_directories(runtime_adapter_tests PRIVATE src)
add_test(NAME recvtable_and_memory_guards COMMAND runtime_adapter_tests)
add_executable(renderer_abi_tests tests/renderer_abi_tests.cpp src/renderer.cpp)
target_include_directories(renderer_abi_tests PRIVATE src)
target_compile_definitions(renderer_abi_tests PRIVATE NOMINMAX)
target_compile_options(renderer_abi_tests PRIVATE /W4 /permissive-)
add_test(NAME renderer_surface_abi COMMAND renderer_abi_tests)
add_executable(plugin_load_tests tests/plugin_load_tests.cpp)
add_test(NAME plugin_sdk_loading COMMAND plugin_load_tests $<TARGET_FILE:NekoExtraHUD>)
add_executable(plugin_hotpath_tests tests/plugin_hotpath_tests.cpp)
target_include_directories(plugin_hotpath_tests PRIVATE src)
target_compile_definitions(plugin_hotpath_tests PRIVATE NOMINMAX)
target_compile_options(plugin_hotpath_tests PRIVATE /W4 /permissive-)
target_link_libraries(plugin_hotpath_tests PRIVATE bcrypt user32)
add_test(NAME plugin_hotpath_gates COMMAND plugin_hotpath_tests)
add_executable(render_preview tests/render_preview.cpp src/layout.cpp src/renderer.cpp)
target_include_directories(render_preview PRIVATE src)
target_compile_definitions(render_preview PRIVATE NOMINMAX)
target_compile_options(render_preview PRIVATE /utf-8)
target_link_libraries(render_preview PRIVATE user32)
